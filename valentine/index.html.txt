<script>

const messages = [
    "Come on jeyjey",
    "Gerçekten mi?",
    "Son şans!",
    "Bunu yapmazsın herhalde?",
    "Kalbimi kırıyorsun :(",
    "Yahu bir düşün!",
    "Hala hayır mı diyorsun?",
    "İnat etme artık :)",
    "Evet'e basmanı bekliyorum!"
];

let clickCount = 0;

function sayNo() {
    const noButton = document.getElementById("no");
    const yesButton = document.getElementById("yes");

    clickCount++;

    if (clickCount < messages.length + 1) {
        noButton.textContent = messages[clickCount - 1];
    }

    if (clickCount >= messages.length + 1) {
        noButton.style.display = "none";
    }

    const newSize = Math.min(17 + clickCount * 4, 65);
    yesButton.style.fontSize = newSize + "px";
    yesButton.style.padding = "14px 27px";
}

function accept() {
    document.getElementById("question").style.display = "none";
    document.getElementById("celebration").style.display = "flex";
}
